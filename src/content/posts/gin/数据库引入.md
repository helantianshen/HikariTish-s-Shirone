## Gorm 引入 MySQL

### 初始化

```go
package main

import (
	"log"
	"os"
	"time"

	"github.com/gin-gonic/gin"

	"gorm.io/driver/mysql"
	"gorm.io/gorm"
	"gorm.io/gorm/logger"
	"gorm.io/gorm/schema"
)

// 全局数据库对象
var DB *gorm.DB

// 示例模型
type User struct {
	ID        uint      `gorm:"primaryKey"`
	Username  string    `gorm:"size:50;unique;not null"`
	Email     string    `gorm:"size:100;unique;not null"`
	Password  string    `gorm:"size:255;not null"`
	Age       int
	CreatedAt time.Time
	UpdatedAt time.Time
}

// 初始化数据库
func InitDB() {

	// MySQL 连接字符串
	dsn := "root:123456@tcp(127.0.0.1:3306)/demo?charset=utf8mb4&parseTime=True&loc=Local"

	// GORM 配置
	config := &gorm.Config{

		// 跳过默认事务（提高性能）
		SkipDefaultTransaction: true,

		// 开启 SQL 预编译
		PrepareStmt: true,

		// 命名策略
		NamingStrategy: schema.NamingStrategy{

			// User -> user
			SingularTable: true,

			// 表名前缀
			TablePrefix: "t_",
		},

		// GORM 日志
		Logger: logger.New(

			log.New(os.Stdout, "\r\n", log.LstdFlags),

			logger.Config{

				// 慢 SQL 阈值
				SlowThreshold: time.Second,

				// 日志级别
				LogLevel: logger.Info,

				// 忽略 RecordNotFound
				IgnoreRecordNotFoundError: true,

				// 彩色输出
				Colorful: true,
			},
		),
	}

	var err error

	// 建立数据库连接
	DB, err = gorm.Open(mysql.Open(dsn), config)
	if err != nil {
		log.Fatal("数据库连接失败：", err)
	}

	// 获取底层 sql.DB
	sqlDB, err := DB.DB()
	if err != nil {
		log.Fatal(err)
	}

	// 数据库连接池配置
	// 最大空闲连接数
	sqlDB.SetMaxIdleConns(10)

	// 最大打开连接数
	sqlDB.SetMaxOpenConns(100)

	// 连接最大生命周期
	sqlDB.SetConnMaxLifetime(time.Hour)

	// 空闲连接最大生命周期
	sqlDB.SetConnMaxIdleTime(30 * time.Minute)

	// 测试数据库连接
	err = sqlDB.Ping()
	if err != nil {
		log.Fatal("数据库 Ping 失败：", err)
	}

	// 自动迁移
	err = DB.AutoMigrate(&User{})
	if err != nil {
		log.Fatal("自动迁移失败：", err)
	}

	log.Println("MySQL 初始化成功")
}

func main() {

	// 初始化数据库
	InitDB()

	// 初始化 Gin
	r := gin.Default()

	// 测试接口
	r.GET("/ping", func(c *gin.Context) {

		c.JSON(200, gin.H{
			"message": "pong",
		})

	})

	// 启动服务
	r.Run(":8080")
}

```

### DSN 参数说明


| 参数              | 说明                             |
| --------------- | ------------------------------ |
| root            | 数据库用户名                         |
| 123456          | 数据库密码                          |
| 127.0.0.1       | 数据库地址                          |
| 3306            | MySQL 默认端口                     |
| demo            | 数据库名称                          |
| charset=utf8mb4 | 使用 UTF-8 字符集，支持 Emoji          |
| parseTime=True  | 自动将 MySQL 的时间类型解析为 `time.Time` |
| loc=Local       | 使用本地时区                         |

## 官方驱动 clickhouse-go/v2 引入 ClickHouse

### 初始化

```go
package main

import (
	"database/sql"
	"log"
	"time"

	"github.com/ClickHouse/clickhouse-go/v2"
	"github.com/gin-gonic/gin"
)

var DB *sql.DB

// 初始化数据库
func InitDB() {

	var err error

	// 建立数据库连接
	DB = clickhouse.OpenDB(&clickhouse.Options{

		// 支持多个节点，官方驱动会自动进行负载均衡和故障切换
		Addr: []string{
		    "192.168.1.10:9000",
		    "192.168.1.11:9000",
		    "192.168.1.12:9000",
		}

		Auth: clickhouse.Auth{
			Database: "default",
			Username: "default",
			Password: "",
		},

		Settings: clickhouse.Settings{
			"max_execution_time": 60,
		},

		DialTimeout: 5 * time.Second,

		Compression: &clickhouse.Compression{
			Method: clickhouse.CompressionLZ4,
		},
	})

	// 数据库连接池配置
	DB.SetMaxIdleConns(10)

	DB.SetMaxOpenConns(100)

	DB.SetConnMaxLifetime(time.Hour)

	DB.SetConnMaxIdleTime(30 * time.Minute)

	// 测试数据库连接
	err = DB.Ping()
	if err != nil {
		log.Fatal("数据库 Ping 失败：", err)
	}

	log.Println("ClickHouse 初始化成功")
}

func main() {

	InitDB()

	r := gin.Default()

	r.GET("/ping", func(c *gin.Context) {

		c.JSON(200, gin.H{
			"message": "pong",
		})

	})

	r.Run(":8080")
}
```

### 配置说明

	Client *mongo.Client
	DB     *mongo.Database
)

// 初始化 MongoDB
func InitMongo() {

	// 创建 Context
	ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
	defer cancel()

	// MongoDB URI
	uri := "mongodb://root:123456@127.0.0.1:27017"

	// 创建 Client 配置
	opts := options.Client().
		ApplyURI(uri).
		SetMaxPoolSize(100).
		SetMinPoolSize(10).
		SetMaxConnIdleTime(30 * time.Minute)

	var err error

	// 建立连接
	Client, err = mongo.Connect(opts)
	if err != nil {
		log.Fatal("MongoDB 连接失败：", err)
	}

	// 测试连接
	err = Client.Ping(ctx, nil)
	if err != nil {
		log.Fatal("MongoDB Ping 失败：", err)
	}

	// 获取数据库
	DB = Client.Database("demo")

	log.Println("MongoDB 初始化成功")
}

func main() {

	InitMongo()

	r := gin.Default()

	r.GET("/ping", func(c *gin.Context) {

		c.JSON(200, gin.H{
			"message": "pong",
		})

	})

	r.Run(":8080")
}
